cmake_minimum_required(VERSION 3.24)
project(android_port_native_dependencies NONE)

include(ExternalProject)

set(ANDROID_PORT_SDL3_REVISION "147a8ee32dbf9ac02f3794964490687b6bbda1bc")
set(ANDROID_PORT_SDL3_IMAGE_REVISION "bec9134a26c7d0f31b36d6083c25296e04cabff5")
set(ANDROID_PORT_FREETYPE_REVISION "42608f77f20749dd6ddc9e0536788eaad70ea4b5")
set(ANDROID_PORT_FMT_REVISION "407c905e45ad75fc29bf0f9bb7c5c2fd3475976f")

set(ANDROID_PORT_SDL3_TTF_REVISION "a1ce3670aec736ecbf0936c43f2f0cc53aa61e5b")
set(ANDROID_PORT_BZIP2_REVISION "6a8690fc8d26c815e798c588f796eabe9d684cf0")

set(ANDROID_PORT_COMMON_CMAKE_ARGS
  -DCMAKE_TOOLCHAIN_FILE=${CMAKE_TOOLCHAIN_FILE}
  -DANDROID_ABI=${ANDROID_ABI}
  -DANDROID_PLATFORM=${ANDROID_PLATFORM}
  -DANDROID_STL=${ANDROID_STL}
  -DCMAKE_BUILD_TYPE=Release
  -DCMAKE_POSITION_INDEPENDENT_CODE=ON
  -DCMAKE_INSTALL_LIBDIR=lib
  -DCMAKE_INSTALL_PREFIX=${CMAKE_INSTALL_PREFIX})

ExternalProject_Add(freetype
  PREFIX "${CMAKE_BINARY_DIR}/freetype"
  SOURCE_DIR "${CMAKE_BINARY_DIR}/sources/freetype"
  GIT_REPOSITORY https://github.com/freetype/freetype.git
  GIT_TAG ${ANDROID_PORT_FREETYPE_REVISION}
  GIT_SHALLOW TRUE
  CMAKE_ARGS
    ${ANDROID_PORT_COMMON_CMAKE_ARGS}
    -DBUILD_SHARED_LIBS=OFF
    -DFT_DISABLE_BZIP2=ON
    -DFT_DISABLE_BROTLI=ON
    -DFT_DISABLE_HARFBUZZ=ON
    -DFT_DISABLE_PNG=ON
    -DFT_DISABLE_ZLIB=ON
  BUILD_BYPRODUCTS "${CMAKE_INSTALL_PREFIX}/lib/libfreetype.a")

ExternalProject_Add(sdl3
  PREFIX "${CMAKE_BINARY_DIR}/sdl3"
  SOURCE_DIR "${CMAKE_BINARY_DIR}/sources/sdl3"
  GIT_REPOSITORY https://github.com/libsdl-org/SDL.git
  GIT_TAG ${ANDROID_PORT_SDL3_REVISION}
  GIT_SHALLOW TRUE
  CMAKE_ARGS
    ${ANDROID_PORT_COMMON_CMAKE_ARGS}
    -DSDL_SHARED=ON
    -DSDL_STATIC=OFF
    -DSDL_INSTALL=ON
    -DSDL_TESTS=OFF
    -DSDL_TEST_LIBRARY=OFF
    -DSDL_EXAMPLES=OFF
    -DSDL_INSTALL_TESTS=OFF
  BUILD_BYPRODUCTS "${CMAKE_INSTALL_PREFIX}/lib/libSDL3.so")

ExternalProject_Add(fmt
  PREFIX "${CMAKE_BINARY_DIR}/fmt"
  SOURCE_DIR "${CMAKE_BINARY_DIR}/sources/fmt"
  GIT_REPOSITORY https://github.com/fmtlib/fmt.git
  GIT_TAG ${ANDROID_PORT_FMT_REVISION}
  GIT_SHALLOW TRUE
  CMAKE_ARGS
    ${ANDROID_PORT_COMMON_CMAKE_ARGS}
    -DBUILD_SHARED_LIBS=OFF
    -DFMT_DOC=OFF
    -DFMT_TEST=OFF
  BUILD_BYPRODUCTS "${CMAKE_INSTALL_PREFIX}/lib/libfmt.a")

ExternalProject_Add(sdl3_image
  PREFIX "${CMAKE_BINARY_DIR}/sdl3_image"
  SOURCE_DIR "${CMAKE_BINARY_DIR}/sources/sdl3_image"
  GIT_REPOSITORY https://github.com/libsdl-org/SDL_image.git
  GIT_TAG ${ANDROID_PORT_SDL3_IMAGE_REVISION}
  GIT_SHALLOW TRUE
  GIT_SUBMODULES external/zlib external/libpng
  GIT_SUBMODULES_RECURSE FALSE
  DEPENDS sdl3
  CMAKE_ARGS
    ${ANDROID_PORT_COMMON_CMAKE_ARGS}
    -DSDL3_DIR=${CMAKE_INSTALL_PREFIX}/lib/cmake/SDL3
    -DBUILD_SHARED_LIBS=OFF
    -DSDLIMAGE_VENDORED=ON
    -DSDLIMAGE_SHARED=OFF
    -DSDLIMAGE_STATIC=ON
    -DSDLIMAGE_INSTALL=ON
    -DSDLIMAGE_TESTS=OFF
    -DSDLIMAGE_SAMPLES=OFF
    -DSDLIMAGE_AVIF=OFF
    -DSDLIMAGE_GIF=OFF
    -DSDLIMAGE_JXL=OFF
    -DSDLIMAGE_TIF=OFF
    -DSDLIMAGE_WEBP=OFF
    -DSDLIMAGE_PNG=ON
    -DSDLIMAGE_PNG_LIBPNG=ON
    -DSDLIMAGE_PNG_SAVE=ON
    -DSDLIMAGE_PNG_SHARED=OFF
  BUILD_BYPRODUCTS "${CMAKE_INSTALL_PREFIX}/lib/libSDL3_image.a")

ExternalProject_Add(sdl3_ttf
  PREFIX "${CMAKE_BINARY_DIR}/sdl3_ttf"
  SOURCE_DIR "${CMAKE_BINARY_DIR}/sources/sdl3_ttf"
  GIT_REPOSITORY https://github.com/libsdl-org/SDL_ttf.git
  GIT_TAG ${ANDROID_PORT_SDL3_TTF_REVISION}
  GIT_SHALLOW TRUE
  GIT_SUBMODULES ""
  DEPENDS sdl3 freetype
  CMAKE_ARGS
    ${ANDROID_PORT_COMMON_CMAKE_ARGS}
    -DCMAKE_PREFIX_PATH=${CMAKE_INSTALL_PREFIX}
    -DSDL3_DIR=${CMAKE_INSTALL_PREFIX}/lib/cmake/SDL3
    -DBUILD_SHARED_LIBS=OFF
    -DFREETYPE_LIBRARY=${CMAKE_INSTALL_PREFIX}/lib/libfreetype.a
    -DFREETYPE_INCLUDE_DIR_ft2build=${CMAKE_INSTALL_PREFIX}/include/freetype2
    -DFREETYPE_INCLUDE_DIR_freetype2=${CMAKE_INSTALL_PREFIX}/include/freetype2
    -DSDLTTF_INSTALL=ON
    -DSDLTTF_VENDORED=OFF
    -DSDLTTF_SAMPLES=OFF
    -DSDLTTF_HARFBUZZ=OFF
    -DSDLTTF_PLUTOSVG=OFF
  BUILD_BYPRODUCTS "${CMAKE_INSTALL_PREFIX}/lib/libSDL3_ttf.a")

ExternalProject_Add(bzip2
  PREFIX "${CMAKE_BINARY_DIR}/bzip2"
  SOURCE_DIR "${CMAKE_BINARY_DIR}/sources/bzip2"
  GIT_REPOSITORY https://sourceware.org/git/bzip2.git
  GIT_TAG ${ANDROID_PORT_BZIP2_REVISION}
  GIT_SHALLOW TRUE
  CONFIGURE_COMMAND ${CMAKE_COMMAND}
    -S "${CMAKE_CURRENT_SOURCE_DIR}/bzip2" -B <BINARY_DIR> -G Ninja
    ${ANDROID_PORT_COMMON_CMAKE_ARGS} -DBZIP2_SOURCE=<SOURCE_DIR>
  BUILD_BYPRODUCTS "${CMAKE_INSTALL_PREFIX}/lib/libbz2.a")

add_custom_target(android_port_native_dependencies
  DEPENDS freetype sdl3 fmt sdl3_image sdl3_ttf bzip2)

configure_file(android-port-dependencies.json.in android-port-dependencies.json @ONLY)
