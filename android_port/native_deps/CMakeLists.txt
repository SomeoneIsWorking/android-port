ExternalProject_Add(openssl
  PREFIX "${CMAKE_BINARY_DIR}/openssl"
  SOURCE_DIR "${CMAKE_BINARY_DIR}/sources/openssl"
  URL https://github.com/openssl/openssl/releases/download/openssl-${ANDROID_PORT_OPENSSL_VERSION}/openssl-${ANDROID_PORT_OPENSSL_VERSION}.tar.gz
  URL_HASH SHA256=${ANDROID_PORT_OPENSSL_SHA256}
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE
  CONFIGURE_COMMAND ${CMAKE_COMMAND} -E env
    ANDROID_NDK_ROOT=${CMAKE_ANDROID_NDK}
    "PATH=${ANDROID_TOOLCHAIN_ROOT}/bin:$ENV{PATH}"
    perl <SOURCE_DIR>/Configure ${ANDROID_PORT_OPENSSL_TARGET}
    -D__ANDROID_API__=${ANDROID_PORT_API} -fPIC
    no-shared no-module no-tests no-docs no-apps
    --prefix=${CMAKE_INSTALL_PREFIX} --libdir=lib
  BUILD_COMMAND ${CMAKE_COMMAND} -E env "PATH=${ANDROID_TOOLCHAIN_ROOT}/bin:$ENV{PATH}"
    make build_libs
  INSTALL_COMMAND ${CMAKE_COMMAND} -E env "PATH=${ANDROID_TOOLCHAIN_ROOT}/bin:$ENV{PATH}"
    make install_dev
  BUILD_BYPRODUCTS
    "${CMAKE_INSTALL_PREFIX}/lib/libssl.a"
    "${CMAKE_INSTALL_PREFIX}/lib/libcrypto.a")

# zlib is the NDK's system libz; no CA store is compiled in, the title supplies one.
ExternalProject_Add(curl
  PREFIX "${CMAKE_BINARY_DIR}/curl"
  SOURCE_DIR "${CMAKE_BINARY_DIR}/sources/curl"
  URL https://github.com/curl/curl/releases/download/${ANDROID_PORT_CURL_TAG}/curl-${ANDROID_PORT_CURL_VERSION}.tar.xz
  URL_HASH SHA256=${ANDROID_PORT_CURL_SHA256}
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE
  DEPENDS openssl
  CMAKE_ARGS
    ${ANDROID_PORT_COMMON_CMAKE_ARGS}
    -DBUILD_SHARED_LIBS=OFF
    -DBUILD_STATIC_LIBS=ON
    -DBUILD_CURL_EXE=OFF
    -DBUILD_TESTING=OFF
    -DBUILD_EXAMPLES=OFF
    -DBUILD_LIBCURL_DOCS=OFF
    -DBUILD_MISC_DOCS=OFF
    -DENABLE_CURL_MANUAL=OFF
    -DCURL_USE_PKGCONFIG=OFF
    -DCURL_USE_OPENSSL=ON
    -DOPENSSL_INCLUDE_DIR=${CMAKE_INSTALL_PREFIX}/include
    -DOPENSSL_SSL_LIBRARY=${CMAKE_INSTALL_PREFIX}/lib/libssl.a
    -DOPENSSL_CRYPTO_LIBRARY=${CMAKE_INSTALL_PREFIX}/lib/libcrypto.a
    -DCURL_ZLIB=ON
    -DCURL_BROTLI=OFF
    -DCURL_ZSTD=OFF
    -DUSE_NGHTTP2=OFF
    -DUSE_LIBIDN2=OFF
    -DCURL_USE_LIBPSL=OFF
    -DCURL_USE_LIBSSH2=OFF
    -DCURL_DISABLE_LDAP=ON
    -DCURL_CA_BUNDLE=none
    -DCURL_CA_PATH=none
  BUILD_BYPRODUCTS "${CMAKE_INSTALL_PREFIX}/lib/libcurl.a")
