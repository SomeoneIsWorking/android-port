cmake_minimum_required(VERSION 3.24)
project(android_port_bzip2 C)

if(NOT EXISTS "${BZIP2_SOURCE}/bzlib.h")
  message(FATAL_ERROR "Pinned bzip2 source is missing: ${BZIP2_SOURCE}")
endif()
set(BZIP2_SOURCES blocksort huffman crctable randtable compress decompress bzlib)
list(TRANSFORM BZIP2_SOURCES PREPEND "${BZIP2_SOURCE}/")
list(TRANSFORM BZIP2_SOURCES APPEND ".c")
add_library(bz2 STATIC ${BZIP2_SOURCES})
target_include_directories(bz2 PRIVATE "${BZIP2_SOURCE}")
target_compile_definitions(bz2 PRIVATE _FILE_OFFSET_BITS=64)
install(TARGETS bz2 ARCHIVE DESTINATION lib)
install(FILES "${BZIP2_SOURCE}/bzlib.h" DESTINATION include)
